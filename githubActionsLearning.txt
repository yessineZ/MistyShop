        - main

    build :
        runs-on : ubuntu-latest
        env :
            API_KEY : ${{secrets.API_KEY}}
        steps :
        - uses : actions/checkout@v2
        - name : Build Docker Image
          run : |
            docker build -t linkedin-frontend /frontend
            docker build -t linkedin-backend /backend

          run : mkdir folder1

         #create a file

          run : ls -l

        runs-on : ubuntu-latest
